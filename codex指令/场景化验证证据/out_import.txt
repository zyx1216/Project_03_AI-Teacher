# C. 归一化函数边界
######################################################################
normalize_gender 取值:
   '男'          -> '男'
   '男生'         -> '男'
   'M'          -> '男'
   'male'       -> '男'
   '1'          -> '男'
   '女'          -> '女'
   '女生'         -> '女'
   'F'          -> '女'
   'female'     -> '女'
   '0'          -> '女'
   ''           -> None
   None         -> None
   '未知'         -> None
normalize_subject_name 取值:
   '数学'         -> '数学'
   '数学成绩'       -> '数学'
   'Math'       -> '数学'
   '数学 成绩'      -> '数学'
   '语 文'        -> '语文'
   '英语分'        -> None
   '科学'         -> None
   '道德与法治'      -> None
   ''           -> None
_to_score 取值（含常见脏值）:
   90           -> 90.0
   '90'         -> 90.0
   '90分'        -> 90.0
   ''           -> None
   None         -> None
   -1           -> None
   '缺考'         -> None
   '未交'         -> None
   '免考'         -> None
   999          -> 999.0
   1000000000.0 -> 1000000000.0
   '90.5'       -> 90.5
